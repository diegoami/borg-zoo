# The Borg's settings (harness/borg.txt). The zoo and ~/borg-play copy this
# file as the run's user/borg.txt; the vendored borg-rs core reads it.
#
# How to read the comments:
# - What the setting does, then what raising or lowering it does.
# - "Change together": settings that only make sense as a set.
# - "Impact": how many of 6 test games (Warrior, Mage, Druid, Priest,
#   Ranger, Blackguard; 50,000 game turns each) play differently when this
#   one value is changed as stated, everything else as shipped (#576). It
#   shows whether the setting is reached early in a game, not whether the
#   change is better or worse: only a zoo run says that. 0 means not reached
#   in these short games (deeper levels, rare monsters, later purchases), not
#   that the setting does nothing.
# - These are the shipped values, not the core's compiled defaults: the
#   compiled core keeps neutral values (docs/CALIBRATION.md), and these
#   lines change them for every game that reads this file (#479). For a
#   switch, 0 turns the feature off.
# - A zoo profile rewrites a line in place; BORG_PARAM_<NAME> in the
#   environment overrides a borg_param_ line, and BORG_PARAM_<NAME>__<CLASS>
#   one class's value (a shell cannot export a name with @). A class's own
#   value lives in harness/borg_<class>.txt, folded into the CLASS FILES
#   block at the end (#520), and wins over the common line here.
#
# Sections: the core, the errand and when to stop; run control; Town; Word
# of Recall; descending and the trip policy; Combat; Healing; Fleeing;
# Cornered; Breeders; Scary monsters; the C Borg's own settings.


### The core, the errand and when to stop ###

# The decision core: "borg-rs" (borg-rs's core, vendored in borg-core/,
# shipped since #479) or "reborn" (this repository's own Borg, not ready
# yet). BORG_CORE in the environment overrides it. harness/run.sh plays the
# core named here when BORG_CORE is unset (#606); BORG_CORE=reborn selects
# the reborn core, which reads its profile, not this file.

borg_reborn_core = borg-rs

# An errand over the core (borg-core/src/shell.rs): "none", "/town",
# "/walk-back", "/explore-back", "/dive N", "/dive-exploring N",
# "/explore-level", "/ladder" (the class's ladder in borg-core/ladders.txt)
# or a plan ("/trips 1*/1,2,1*@3/..."). BORG_ERRAND overrides it; in the
# game, ^Z / types one. "none" lets the core play as itself.

borg_reborn_errand = none

# Hand control back to the player when the danger where the Borg stands
# reaches this percent of its current HP: "off", N (1-1000), or "flee"
# (100: one turn could kill it). Lower stops it earlier. BORG_STOPSCARED
# overrides it.

borg_reborn_stopscared = off

# Stop at this dungeon level or character level (out of range = never), go
# no deeper than borg_no_deeper, and stop after killing Morgoth
# (borg_stop_king). Lowering borg_no_deeper caps the dive: the Borg stays at
# or above that level and gains less experience.

borg_stop_dlevel = 128
borg_stop_clevel = 51
borg_no_deeper = 127
borg_stop_king = TRUE

# Stop when the game rings the bell, which generally means an error; the
# message log then lists the last 50 keypresses. For debugging only.

borg_stop_on_bell = FALSE

### Town ###

# Shopping, selling and the town planner. With borg_town_plan = 1 (shipped)

# 2 x borg_town_stair_turns x depth (or borg_town_trip_turns, if not 0). The
# longer and deeper the forecast trip, the more light, food, healing and
# escapes the planner buys.
# Change together: depth_gain, explore_turns and stair_turns set the trip's
# length; target_depth and trip_turns replace them with fixed values.
# Impact (games of 6 that play differently with the value changed to):
# town_depth_gain 2: 6; town_explore_turns 2000: 6; town_stair_turns 100: 5;

borg_town_depth_gain = 1

borg_town_target_depth = 0
borg_town_trip_turns = 0

# Supplies per trip: one light source lasts borg_town_light_turns turns and
# one ration borg_town_food_turns. Lower buys more light and food.
# Impact (games of 6 that play differently with the value changed to):
# town_light_turns 10000: 0; town_food_turns 6000: 0.

borg_town_light_turns = 5000
borg_town_food_turns = 3000

# Exposure: the trip's turns x depth x borg_town_risk / 1000 / character
# level sizes the healing and escapes wanted (1000 = one max-HP of healing

# escapes and less of everything else. borg_town_power_scale is how much
# better gear is worth per level gained; higher spends more on gear.
# Impact (games of 6 that play differently with the value changed to):

borg_town_risk = 1000

# The weight of each need, against light at 100. Each weight scales how much

# - bleeding, poison: cures for cuts and poison;
# - escape: Phase Door, teleport and similar;
# - knowledge: buying one of each unknown flavour to learn it; forced to 0

#   sell unknown items instead), so this line applies only with the policy
#   off;
# - power: better gear;
# - trip: Word of Recall (the walk it saves);
# - attack, magic, support, restore: missiles and attack devices, magic
#   devices, buff potions and scrolls, stat restoration;
# - ranger_missiles: arrows kept for a Ranger (0 = none).
# borg_town_attack_turns, support_turns and ranger_missile_turns are the
# forecast turns per use: lower means more uses per trip, so more bought.
